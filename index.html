import math import tkinter as tk from tkinter import filedialog, messagebox, ttk class ImageToAsciiApp: def __init__(self, root): self.root = root self.root.title("Контур и ASCII Генератор") self.root.geometry("900x700") self.img_pixels = None self.img_width = 0 self.img_height = 0 self.setup_ui() def setup_ui(self): # Верхняя панель управления control_frame = ttk.Frame(self.root, padding=10) control_frame.pack(fill=tk.X) btn_load = ttk.Button( control_frame, text="Загрузить фото", command=self.load_image ) btn_load.pack(side=tk.LEFT, padx=5) ttk.Label(control_frame, text="Режим:").pack( side=tk.LEFT, padx=(15, 2) ) self.mode_var = tk.StringVar(value="contour") mode_cb = ttk.Combobox( control_frame, textvariable=self.mode_var, values=["Контур точками", "ASCII из символов"], state="readonly", width=18, ) mode_cb.current(0) mode_cb.pack(side=tk.LEFT, padx=5) mode_cb.bind("<>", lambda e: self.process_image()) ttk.Label(control_frame, text="Ширина текста:").pack( side=tk.LEFT, padx=(15, 2) ) self.width_scale = ttk.Scale( control_frame, from_=40, to=200, value=100, command=lambda e: self.process_image(), ) self.width_scale.pack(side=tk.LEFT, padx=5) ttk.Label(control_frame, text="Порог/Чувствительность:").pack( side=tk.LEFT, padx=(15, 2) ) self.thresh_scale = ttk.Scale( control_frame, from_=10, to=150, value=50, command=lambda e: self.process_image(), ) self.thresh_scale.pack(side=tk.LEFT, padx=5) btn_copy = ttk.Button( control_frame, text="Скопировать текст", command=self.copy_to_clipboard ) btn_copy.pack(side=tk.RIGHT, padx=5) # Область вывода текста text_frame = ttk.Frame(self.root, padding=10) text_frame.pack(fill=tk.BOTH, expand=True) self.text_area = tk.Text( text_frame, font=("Courier", 6), wrap=tk.NONE, bg="black", fg="white", ) scrollbar_y = ttk.Scrollbar( text_frame, orient=tk.VERTICAL, command=self.text_area.yview ) scrollbar_x = ttk.Scrollbar( text_frame, orient=tk.HORIZONTAL, command=self.text_area.xview ) self.text_area.configure( yscrollcommand=scrollbar_y.set, xscrollcommand=scrollbar_x.set ) scrollbar_y.pack(side=tk.RIGHT, fill=tk.Y) scrollbar_x.pack(side=tk.BOTTOM, fill=tk.X) self.text_area.pack(fill=tk.BOTH, expand=True) def load_image(self): file_path = filedialog.askopenfilename( filetypes=[ ("Изображения", "*.png *.gif *.ppm"), ("Все файлы", "*.*"), ] ) if not file_path: return try: # Используем встроенный PhotoImage (поддерживает PNG, GIF, PPM) img = tk.PhotoImage(file="file_path") self.img_width = img.width() self.img_height = img.height() # Считываем RGB пиксели self.img_pixels = [] for y in range(self.img_height): row = [] for x in range(self.img_width): r, g, b = img.get(x, y) # Перевод в отрезки яркости (Grayscale) gray = int(0.299 * r + 0.587 * g + 0.114 * b) row.append(gray) self.img_pixels.append(row) self.process_image() except Exception as e: messagebox.showerror( "Ошибка", f"Не удалось загрузить изображение: {e}" ) def process_image(self): if not self.img_pixels: return target_width = int(self.width_scale.get()) # Символы в моноширинном шрифте обычно выше, чем шире (аспект около 0.55) aspect_ratio = 0.55 target_height = int( (self.img_height / self.img_width) * target_width * aspect_ratio ) if target_height <= 0 or target_width <= 0: return # Масштабирование изображения методом ближайшего соседа resized = [] for y in range(target_height): row = [] orig_y = int(y * self.img_height / target_height) for x in range(target_width): orig_x = int(x * self.img_width / target_width) row.append(self.img_pixels[orig_y][orig_x]) resized.append(row) mode = self.mode_var.get() threshold = self.thresh_scale.get() result_lines = [] if mode == "Контур точками": # Вычисление границ с использованием оператора Собеля for y in range(target_height): line = [] for x in range(target_width): if ( y == 0 or y == target_height - 1 or x == 0 or x == target_width - 1 ): line.append(" ") continue # Градиенты по X и Y gx = ( -resized[y - 1][x - 1] + resized[y - 1][x + 1] - 2 * resized[y][x - 1] + 2 * resized[y][x + 1] - resized[y + 1][x - 1] + resized[y + 1][x + 1] ) gy = ( -resized[y - 1][x - 1] - 2 * resized[y - 1][x] - resized[y - 1][x + 1] + resized[y + 1][x - 1] + 2 * resized[y + 1][x] + resized[y + 1][x + 1] ) magnitude = math.sqrt(gx * gx + gy * gy) if magnitude > threshold * 3: line.append(".") else: line.append(" ") result_lines.append("".join(line)) else: #
